    13→/**
    14→ * QUANTUM WAVE PACKET — the one-dimensional time-dependent Schrödinger
    15→ * equation, solved live on a grid.
    16→ *
    17→ * i ∂ψ/∂t = [ −(1/2) ∂²/∂x² + V(x) ] ψ      (natural units ħ = m = 1)
    18→ *
    19→ * The integrator is Crank–Nicolson — (I + i·dt/2·H)ψⁿ⁺¹ = (I − i·dt/2·H)ψⁿ —
    20→ * unconditionally stable and unitary, so probability is conserved exactly
    21→ * (up to round-off) until the absorbing potential at the domain edges
    22→ * deliberately removes outgoing waves, mimicking an infinite domain.
    23→ * The left-side tridiagonal system is factorised once per parameter change
    24→ * (Thomas algorithm over complex re/im arrays); each step is then O(N).
    25→ *
    26→ * Approximations: single particle, non-relativistic, potential barrier
    27→ * static in time. The transmitted/reflected readouts are packet
    28→ * measurements; the comparison figure is the plane-wave formula.
    29→ */
    30→
    31→const N = 768;          // grid points
    32→const X_MIN = -24;      // domain
    33→const X_MAX = 24;
    34→const DX = (X_MAX - X_MIN) / N;
    35→const DT = 0.005;       // Crank–Nicolson is unconditionally stable
    36→const X0 = -14;          // packet start
    37→const SIGMA = 1.2;      // position spread
    38→const CAP_WIDTH = 6;    // absorbing ramp width (each edge)
    39→const E_MIN = 0.4, E_MAX = 3.0;
    40→const V0_MIN = 0.2, V0_MAX = 4.0;
    41→const W_MIN = 0.4, W_MAX = 4.0;

    49→function hexToRgb(hex: string): [number, number, number] {
    50→  const m = hex.replace("#", "");
    51→  const v = m.length === 3 ? m.split("").map((c) => c + c).join("") : m;
    52→  const n = parseInt(v, 16);
    53→  return [(n >> 16) & 255, (n >> 8) & 255, n & 255];
    54→}

    62→/* ---------- plane-wave transmission through a rectangular barrier ---------- */
    63→
    64→function planeWaveT(E: number, V0: number, w: number): number {
    65→  const dE = E - V0;
    66→  if (Math.abs(dE) < 1e-6) {
    67→    // E → V₀ limit: T → 1/(1 + V₀w²/2)
    68→    return 1 / (1 + (V0 * w * w) / 2);
    69→  }
    70→  if (dE < 0) {
    71→    const kappa = Math.sqrt(2 * Math.max(0, V0 - E));
    72→    const s = Math.sinh(kappa * w);
    73→    return 1 / (1 + (V0 * V0 * s * s) / (4 * E * (V0 - E)));
    74→  }
    75→  const k2 = Math.sqrt(2 * dE);
    76→  const s = Math.sin(k2 * w);
    77→  return 1 / (1 + (V0 * V0 * s * s) / (4 * E * dE));
    78→}

    80→export function WaveSim() {
    81→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
    82→
    83→  // null = automatic: paused when the visitor prefers reduced motion
    84→  const [playingOverride, setPlayingOverride] = useState<boolean | null>(null);
    85→  const reduceMotion = useSyncExternalStore(
    86→    subscribeReduceMotion,
    87→    getReduceMotion,
    88→    getServerReduceMotion
    89→  );
    90→  const playing = playingOverride ?? !reduceMotion;
    91→
    92→  const [energy, setEnergy] = useState(1.4);
    93→  const [barrierV, setBarrierV] = useState(1.6);
    94→  const [barrierW, setBarrierW] = useState(1.6);
    95→  const [speed, setSpeed] = useState(5);
    96→
    97→  /* ---------- simulation state (refs; the loop never re-renders React) ---------- */
    98→
    99→  const re = useRef(new Float64Array(N));
   100→  const im = useRef(new Float64Array(N));
   101→  const vPot = useRef(new Float64Array(N));   // real potential
   102→  const gamma = useRef(new Float64Array(N));  // absorbing potential
   103→
   104→  // Crank–Nicolson factorisation of A = I + i·dt/2·H (complex, stored as re/im)
   105→  const diagRe = useRef(new Float64Array(N));
   106→  const diagIm = useRef(new Float64Array(N));
   107→  const cPrimeRe = useRef(new Float64Array(N));
   108→  const cPrimeIm = useRef(new Float64Array(N));
   109→  const pRe = useRef(new Float64Array(N));
   110→  const pIm = useRef(new Float64Array(N));
   111→  const rhsRe = useRef(new Float64Array(N));
   112→  const rhsIm = useRef(new Float64Array(N));
   113→  const dRe = useRef(new Float64Array(N));
   114→  const dIm = useRef(new Float64Array(N));
   115→
   116→  // off-diagonal of H (real, constant): −1/(2dx²)
   117→  const H_OFF = -1 / (2 * DX * DX);
   118→
   119→  const tRef = useRef(0);
   120→  const yMaxRef = useRef(0.35);   // smoothed |ψ|² axis scale
   121→  const doneRef = useRef(false);  // packet fully absorbed
   122→  const readout = useRef<{ t?: HTMLSpanElement | null; x?: HTMLSpanElement | null; R?: HTMLSpanElement | null; T?: HTMLSpanElement | null; A?: HTMLSpanElement | null }>({});
   123→
   124→  const energyRef = useRef(energy);
   125→  const v0Ref = useRef(barrierV);
   126→  const wRef = useRef(barrierW);
   127→  const speedRef = useRef(speed);
   128→  const playingRef = useRef(playing);
   129→  const visibleRef = useRef(true);
   130→  const rafRef = useRef(0);
   131→  const sizeRef = useRef({ w: 0, h: 0, dpr: 1 });
   132→
   133→  /* ---------- build the system for current V₀, w, E ---------- */
   134→
   135→  const buildSystem = useCallback(() => {
   136→    const V0 = v0Ref.current;
   137→    const w = wRef.current;
   138→    const E = energyRef.current;
   139→    const gMax = Math.min(0.35, Math.max(0.03, 0.05 * Math.max(E, V0)));
   140→
   141→    for (let j = 0; j < N; j++) {
   142→      const x = X_MIN + (j + 0.5) * DX;
   143→      vPot.current[j] = Math.abs(x) < w / 2 ? V0 : 0;
   144→
   145→      // absorbing ramp near both edges
   146→      let g = 0;
   147→      if (x < X_MIN + CAP_WIDTH) g = (X_MIN + CAP_WIDTH - x) / CAP_WIDTH;
   148→      else if (x > X_MAX - CAP_WIDTH) g = (x - (X_MAX - CAP_WIDTH)) / CAP_WIDTH;
   149→      gamma.current[j] = gMax * g * g;
   150→
   151→      // H diagonal: 1/dx² + V − iγ  →  A diagonal: 1 + i·(dt/2)·(that)
   152→      //   1 + i·(dt/2)·(hRe + i·hIm) = [1 − (dt/2)·hIm] + i·(dt/2)·hRe
   153→      const hDiagRe = 1 / (DX * DX) + vPot.current[j];
   154→      const hDiagIm = -gamma.current[j];
   155→      diagRe.current[j] = 1 - (DT / 2) * hDiagIm;
   156→      diagIm.current[j] = (DT / 2) * hDiagRe;
   157→    }
   158→
   159→    // Dirichlet edges: row 0 and N−1 become identity so ψ stays 0 there
   160→    diagRe.current[0] = 1; diagIm.current[0] = 0;
   161→    diagRe.current[N - 1] = 1; diagIm.current[N - 1] = 0;
   162→
   163→    // Thomas forward elimination (complex):
   164→    //   p₀ = b₀,  c'₀ = 0 (identity row)
   165→    //   pⱼ = bⱼ − a·c'ⱼ₋₁,  c'ⱼ = c/pⱼ
   166→    // sub/super of A are purely imaginary: i·offIm (offIm = (dt/2)·H_OFF)
   167→    const offIm = (DT / 2) * H_OFF;
   168→    for (let j = 0; j < N; j++) {
   169→      const isEdge = j === 0 || j === N - 1;
   170→      const aIm = isEdge ? 0 : offIm;        // sub-diagonal of row j (identity on edges)
   171→      const cUse = isEdge ? 0 : offIm;       // super-diagonal of row j
   172→
   173→      let pr: number, pi: number;
   174→      if (j === 0) {
   175→        pr = diagRe.current[0];
   176→        pi = diagIm.current[0];
   177→      } else {
   178→        // pⱼ = bⱼ − a·c'ⱼ₋₁  with a = 0 + i·aIm:
   179→        //   (i·aIm)·(c'r + i·c'i) = −aIm·c'i + i·aIm·c'r
   180→        const acRe = -aIm * cPrimeIm.current[j - 1];
   181→        const acIm = aIm * cPrimeRe.current[j - 1];
   182→        pr = diagRe.current[j] - acRe;
   183→        pi = diagIm.current[j] - acIm;
   184→      }
   185→      pRe.current[j] = pr;
   186→      pIm.current[j] = pi;
   187→
   188→      if (j === N - 1) {
   189→        cPrimeRe.current[j] = 0;
   190→        cPrimeIm.current[j] = 0;
   191→      } else {
   192→        // c'ⱼ = c/pⱼ with c = 0 + i·cUse (purely imaginary):
   193→        //   (i·cUse)/(pr + i·pi) = [cUse·pi + i·cUse·pr]/den
   194→        const den = pr * pr + pi * pi;
   195→        cPrimeRe.current[j] = (cUse * pi) / den;
   196→        cPrimeIm.current[j] = (cUse * pr) / den;
   197→      }
   198→    }
   199→  }, []);
   200→
   201→  /* ---------- initial Gaussian packet ---------- */
   202→
   203→  const resetSim = useCallback(() => {
   204→    const k0 = Math.sqrt(2 * energyRef.current);
   205→    let norm = 0;
   206→    for (let j = 0; j < N; j++) {
   207→      const x = X_MIN + (j + 0.5) * DX;
   208→      const env = Math.exp(-((x - X0) * (x - X0)) / (4 * SIGMA * SIGMA));
   209→      re.current[j] = env * Math.cos(k0 * x);
   210→      im.current[j] = env * Math.sin(k0 * x);
   211→      norm += (re.current[j] ** 2 + im.current[j] ** 2) * DX;
   212→    }
   213→    const f = 1 / Math.sqrt(norm);
   214→    for (let j = 0; j < N; j++) {
   215→      re.current[j] *= f;
   216→      im.current[j] *= f;
   217→    }
   218→    tRef.current = 0;
   219→    doneRef.current = false;
   220→    yMaxRef.current = Math.max(0.12, (1 / (SIGMA * Math.sqrt(2 * Math.PI))) * 1.15);
   221→  }, []);
   222→
   223→  /* ---------- one Crank–Nicolson step ---------- */
   224→
   225→  const step = useCallback(() => {
   226→    const reA = re.current, imA = im.current;
   227→    const vp = vPot.current, gm = gamma.current;
   228→    const rRe = rhsRe.current, rIm = rhsIm.current;
   229→    const dt2 = DT / 2;
   230→
   231→    // rhs = ψⁿ − i·(dt/2)·Hψⁿ
   232→    for (let j = 1; j < N - 1; j++) {
   233→      // (Hψ)ⱼ = H_OFF·(ψⱼ₋₁ + ψⱼ₊₁) + (1/dx² + Vⱼ − iγⱼ)·ψⱼ
   234→      const hr = H_OFF * (reA[j - 1] + reA[j + 1]) + (1 / (DX * DX) + vp[j]) * reA[j] + gm[j] * imA[j];
   235→      const hi = H_OFF * (imA[j - 1] + imA[j + 1]) + (1 / (DX * DX) + vp[j]) * imA[j] - gm[j] * reA[j];
   236→      // −i·(dt/2)·(hr + i·hi) = (dt/2)·hi − i·(dt/2)·hr
   237→      rRe[j] = reA[j] + dt2 * hi;
   238→      rIm[j] = imA[j] - dt2 * hr;
   239→    }
   240→    rRe[0] = 0; rIm[0] = 0;
   241→    rRe[N - 1] = 0; rIm[N - 1] = 0;
   242→
   243→    const offIm = dt2 * H_OFF;
   244→
   245→    // forward: d'₀ = rhs₀/p₀;  d'ⱼ = (rhsⱼ − α·d'ⱼ₋₁)/pⱼ
   246→    {
   247→      const pr = pRe.current[0], pi = pIm.current[0];
   248→      const den = pr * pr + pi * pi;
   249→      dRe.current[0] = (rRe[0] * pr + rIm[0] * pi) / den;
   250→      dIm.current[0] = (rIm[0] * pr - rRe[0] * pi) / den;
   251→    }
   252→    for (let j = 1; j < N; j++) {
   253→      const pr = pRe.current[j], pi = pIm.current[j];
   254→      // α·d'ⱼ₋₁ — α = i·offIm (purely imaginary): i·offIm·(dr + i·di) = −offIm·di + i·offIm·dr
   255→      const aDr = -offIm * dIm.current[j - 1];
   256→      const aDi = offIm * dRe.current[j - 1];
   257→      const isLast = j === N - 1;
   258→      const nr = rRe[j] - (isLast ? 0 : aDr);
   259→      const ni = rIm[j] - (isLast ? 0 : aDi);
   260→      const den = pr * pr + pi * pi;
   261→      dRe.current[j] = (nr * pr + ni * pi) / den;
   262→      dIm.current[j] = (ni * pr - nr * pi) / den;
   263→    }
   264→
   265→    // back-substitution: ψⱼ = d'ⱼ − c'ⱼ·ψⱼ₊₁
   266→    reA[N - 1] = dRe.current[N - 1];
   267→    imA[N - 1] = dIm.current[N - 1];
   268→    for (let j = N - 2; j >= 1; j--) {
   269→      const cr = cPrimeRe.current[j], ci = cPrimeIm.current[j];
   270→      const xr = reA[j + 1], xi = imA[j + 1];
   271→      reA[j] = dRe.current[j] - (cr * xr - ci * xi);
   272→      imA[j] = dIm.current[j] - (cr * xi + ci * xr);
   273→    }
   274→    reA[0] = 0; imA[0] = 0;
   275→
   276→    tRef.current += DT;
   277→  }, []);
   278→
   279→  /* ---------- measured quantities ---------- */
   280→
   281→  const measure = useCallback(() => {
   282→    let total = 0, left = 0, right = 0, xs = 0;
   283→    const w = wRef.current;
   284→    for (let j = 0; j < N; j++) {
   285→      const p = (re.current[j] ** 2 + im.current[j] ** 2) * DX;
   286→      total += p;
   287→      const x = X_MIN + (j + 0.5) * DX;
   288→      if (x < -(w / 2 + 1)) left += p;
   289→      else if (x > w / 2 + 1) right += p;
   290→      xs += x * p;
   291→    }
   292→    return { total, left, right, xMean: total > 1e-12 ? xs / total : 0 };
   293→  }, []);
   294→
   295→  /* ---------- drawing ---------- */
   296→
   297→  const draw = useCallback(() => {
   298→    const canvas = canvasRef.current;
   299→    const ctx = canvas?.getContext("2d");
   300→    if (!canvas || !ctx) return;
   301→    const { w, h, dpr } = sizeRef.current;
   302→    if (!w || !h) return;
   303→
   304→    const photon = hexToRgb(cssVar("--photon"));
   305→    const nebula = hexToRgb(cssVar("--nebula"));
   306→    const dust = hexToRgb(cssVar("--dust"));
   307→    const bg = hexToRgb(cssVar("--background"));
   308→
   309→    const ML = 46, MR = 14, MT = 18, MB = 34;
   310→    const plotW = w - ML - MR;
   311→    const plotH = h - MT - MB;
   312→
   313→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   314→    ctx.fillStyle = `rgb(${bg[0]}, ${bg[1]}, ${bg[2]})`;
   315→    ctx.fillRect(0, 0, w, h);
   316→
   317→    // dynamic y-scale (smoothed so tunneling tails stay visible)
   318→    let frameMax = 0.02;
   319→    for (let j = 0; j < N; j++) {
   320→      const p = re.current[j] ** 2 + im.current[j] ** 2;
   321→      if (p > frameMax) frameMax = p;
   322→    }
   323→    yMaxRef.current += (frameMax * 1.18 - yMaxRef.current) * 0.09;
   324→    const yMax = Math.max(0.02, yMaxRef.current);
   325→
   326→    const X = (x: number) => ML + ((x - X_MIN) / (X_MAX - X_MIN)) * plotW;
   327→    const Y = (p: number) => MT + plotH * (1 - Math.min(1, p / yMax));
   328→
   329→    // barrier band
   330→    const wB = wRef.current;
   331→    const bx0 = X(-wB / 2), bx1 = X(wB / 2);
   332→    ctx.fillStyle = rgba(nebula, 0.1);
   333→    ctx.fillRect(bx0, MT, bx1 - bx0, plotH);
   334→    ctx.strokeStyle = rgba(nebula, 0.55);
   335→    ctx.lineWidth = 1;
   336→    ctx.strokeRect(bx0, MT, bx1 - bx0, plotH);
   337→    ctx.fillStyle = rgba(nebula, 0.9);
   338→    ctx.font = "10px ui-monospace, monospace";
   339→    ctx.textAlign = "center";
   340→    ctx.fillText("V₀", (bx0 + bx1) / 2, MT + 12);
   341→
   342→    // axes
   343→    ctx.strokeStyle = rgba(dust, 0.4);
   344→    ctx.lineWidth = 1;
   345→    ctx.beginPath();
   346→    ctx.moveTo(ML, MT + plotH + 0.5);
   347→    ctx.lineTo(ML + plotW, MT + plotH + 0.5);
   348→    ctx.moveTo(ML + 0.5, MT);
   349→    ctx.lineTo(ML + 0.5, MT + plotH);
   350→    ctx.stroke();
   351→
   352→    // x ticks
   353→    ctx.fillStyle = rgba(dust, 0.85);
   354→    ctx.textAlign = "center";
   355→    for (const tx of [-20, -10, 0, 10, 20]) {
   356→      const px = X(tx);
   357→      ctx.beginPath();
   358→      ctx.moveTo(px, MT + plotH);
   359→      ctx.lineTo(px, MT + plotH + 4);
   360→      ctx.strokeStyle = rgba(dust, 0.4);
   361→      ctx.stroke();
   362→      ctx.fillText(String(tx), px, MT + plotH + 16);
   363→    }
   364→
   365→    // y-axis label
   366→    ctx.textAlign = "left";
   367→    ctx.fillStyle = rgba(dust, 0.75);
   368→    ctx.fillText("|ψ|²", 8, MT + 4);
   369→
   370→    // energy annotation
   371→    ctx.textAlign = "right";
   372→    const E = energyRef.current, V0 = v0Ref.current;
   373→    ctx.fillText(`E = ${E.toFixed(2)} · V₀ = ${V0.toFixed(2)} · E/V₀ = ${(E / Math.max(V0, 1e-9)).toFixed(2)}`, ML + plotW, MT - 6 + 10);
   374→
   375→    // |ψ|² filled curve
   376→    ctx.beginPath();
   377→    ctx.moveTo(X(X_MIN), Y(0));
   378→    for (let j = 0; j < N; j++) {
   379→      const x = X_MIN + (j + 0.5) * DX;
   380→      ctx.lineTo(X(x), Y(re.current[j] ** 2 + im.current[j] ** 2));
   381→    }
   382→    ctx.lineTo(X(X_MAX - DX / 2), Y(0));
   383→    ctx.closePath();
   384→    ctx.fillStyle = rgba(photon, 0.14);
   385→    ctx.fill();
   386→
   387→    ctx.beginPath();
   388→    for (let j = 0; j < N; j++) {
   389→      const x = X_MIN + (j + 0.5) * DX;
   390→      const px = X(x), py = Y(re.current[j] ** 2 + im.current[j] ** 2);
   391→      if (j === 0) ctx.moveTo(px, py);
   392→      else ctx.lineTo(px, py);
   393→    }
   394→    ctx.strokeStyle = rgba(photon, 0.92);
   395→    ctx.lineWidth = 1.6;
   396→    ctx.stroke();
   397→
   398→    // Re ψ (thin) — only when visible against the scale
   399→    const reMax = yMax * 0.04;
   400→    let showRe = false;
   401→    for (let j = 0; j < N; j++) if (Math.abs(re.current[j]) > reMax) { showRe = true; break; }
   402→    if (showRe) {
   403→      ctx.beginPath();
   404→      for (let j = 0; j < N; j++) {
   405→        const x = X_MIN + (j + 0.5) * DX;
   406→        const py = MT + plotH * (1 - Math.max(-1, Math.min(1, (re.current[j] / Math.sqrt(yMax)) * 0.28)));
   407→        if (j === 0) ctx.moveTo(X(x), py);
   408→        else ctx.lineTo(X(x), py);
   409→      }
   410→      ctx.strokeStyle = rgba(dust, 0.5);
   411→      ctx.lineWidth = 1;
   412→      ctx.stroke();
   413→    }
   414→
   415→    // done note
   416→    if (doneRef.current) {
   417→      ctx.textAlign = "center";
   418→      ctx.fillStyle = rgba(dust, 0.9);
   419→      ctx.fillText("packet absorbed — press R to reset", ML + plotW / 2, MT + 16);
   420→    }
   421→  }, []);
   422→
   423→  /* ---------- lifecycle ---------- */
   424→
   425→  useEffect(() => {
   426→    const canvas = canvasRef.current;
   427→    if (!canvas) return;
   428→
   429→    const resize = () => {
   430→      const parent = canvas.parentElement;
   431→      if (!parent) return;
   432→      const rect = parent.getBoundingClientRect();
   433→      if (rect.width < 2 || rect.height < 2) return;
   434→      const dpr = Math.min(window.devicePixelRatio || 1, 2);
   435→      sizeRef.current = { w: rect.width, h: rect.height, dpr };
   436→      canvas.width = Math.round(rect.width * dpr);
   437→      canvas.height = Math.round(rect.height * dpr);
   438→    };
   439→
   440→    buildSystem();
   441→    resetSim();
   442→    resize();
   443→
   444→    const ro = new ResizeObserver(resize);
   445→    ro.observe(canvas.parentElement ?? canvas);
   446→
   447→    const io = new IntersectionObserver(
   448→      (entries) => { visibleRef.current = entries[0]?.isIntersecting ?? true; },
   449→      { threshold: 0.05 }
   450→    );
   451→    io.observe(canvas);
   452→
   453→    const onVisibility = () => { visibleRef.current = document.visibilityState === "visible"; };
   454→    document.addEventListener("visibilitychange", onVisibility);
