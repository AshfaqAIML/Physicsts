   102→   102→function hexToRgb(hex: string): [number, number, number] {
   103→   103→  const m = hex.replace("#", "");
   104→   104→  const v = m.length === 3 ? m.split("").map((c) => c + c).join("") : m;
   105→   105→  const n = parseInt(v, 16);
   106→   106→  return [(n >> 16) & 255, (n >> 8) & 255, n & 255];
   107→   107→}

   139→   139→  const vRef = useRef<Vec>({ x: 0, y: 1 });
   140→   140→  const e0Ref = useRef(0); // initial specific energy
   141→   141→  const orbitRef = useRef({ h: 1, e: 0, a: 1, bound: true, thetaP: Math.PI });
   142→   142→  const scaleRef = useRef(1); // px per sim unit
   143→   143→  const ratioRef = useRef(massRatio);
   144→   144→  const speedRef = useRef(speed);
   145→   145→  const analyticRef = useRef(showAnalytic);

   149→   149→  const rafRef = useRef(0);
   150→   150→  const sizeRef = useRef({ w: 0, h: 0, dpr: 1 });
   151→   151→
   152→   152→  const readE = useRef<HTMLSpanElement | null>(null);
   153→   153→  const readA = useRef<HTMLSpanElement | null>(null);

   155→   155→  const readT = useRef<HTMLSpanElement | null>(null);

   158→   158→  /* ---------- initial conditions & analytic conic ---------- */
   159→   159→
   160→   160→  const resetSim = useCallback((f: number, keepLensPosition = false) => {
   161→   161→    const { w, h } = sizeRef.current;
   162→   162→    rRef.current = { x: 1, y: 0 };
   163→   163→    vRef.current = { x: 0, y: f }; // tangential launch; v_circ = 1 at r = 1
   164→   164→
   165→   165→    const eps = (f * f) / 2 - 1;
   166→   166→    const hAng = f;
   167→   167→    const e = Math.sqrt(Math.max(0, 1 + 2 * eps * hAng * hAng));
   168→   168→    const bound = eps < 0;
   169→   169→    const a = bound ? -1 / (2 * eps) : Infinity;
   170→   170→    const thetaP = f >= 1 ? 0 : Math.PI; // periapsis direction
   171→   171→
   172→   172→    e0Ref.current = eps;
   173→   173→    orbitRef.current = { h: hAng, e, a, bound, thetaP };
   174→   174→
   175→   175→    // Fit the view to the whole conic
   176→   176→    const rMax = bound ? a * (1 + e) : 1.6;
   177→   177→    const minDim = Math.min(w, h);
   178→   178→    scaleRef.current = (0.42 * minDim) / Math.max(rMax, 1);
   179→   179→
   180→   180→    if (!keepLensPosition) {
   181→   181→      const canvas = canvasRef.current;
   182→   182→      const ctx = canvas?.getContext("2d");
   183→   183→      if (canvas && ctx) {
   184→   184→        ctx.save();
   185→   185→        ctx.setTransform(1, 0, 0, 1, 0, 0);
   186→   186→        ctx.clearRect(0, 0, canvas.width, canvas.height);
   187→   187→        ctx.restore();
   188→   188→      }
   189→   189→    }
   190→   190→  }, []);
   191→   191→
   192→   192→  /* ---------- integration & drawing ---------- */
   193→   193→
   194→   194→  const step = () => {
   195→   195→    const r = rRef.current;
   196→   196→    const v = vRef.current;
   197→   197→
   198→   198→    const r1 = Math.hypot(r.x, r.y);
   199→   199→    const k1 = -1 / (r1 * r1 * r1); // r̈ = −r/|r|³ (GM = 1)
   200→   200→    const ax1 = k1 * r.x;
   201→   201→    const ay1 = k1 * r.y;
   202→   202→
   203→   203→    r.x += v.x * DT + 0.5 * ax1 * DT * DT;
   204→   204→    r.y += v.y * DT + 0.5 * ay1 * DT * DT;
   205→   205→
   206→   206→    const r2 = Math.hypot(r.x, r.y);
   207→   207→    const k2 = -1 / (r2 * r2 * r2);
   208→   208→    const ax2 = k2 * r.x;
   209→   209→    const ay2 = k2 * r.y;
   210→   210→
   211→   211→    v.x += 0.5 * (ax1 + ax2) * DT;
   212→   212→    v.y += 0.5 * (ay1 + ay2) * DT;
   213→   213→  };
   214→   214→
   215→   215→  const draw = useCallback(() => {
   216→   216→    const canvas = canvasRef.current;
   217→   217→    const ctx = canvas?.getContext("2d");
   218→   218→    if (!canvas || !ctx) return;
   219→   219→    const { w, h, dpr } = sizeRef.current;
   220→   220→    if (!w || !h) return;
   221→   221→
   222→   222→    const dark = document.documentElement.classList.contains("dark");
   223→   223→    const bg = hexToRgb(cssVar("--background"));
   224→   224→    const photon = hexToRgb(cssVar("--photon"));
   225→   225→    const dust = hexToRgb(cssVar("--dust"));
   226→   226→    const star: [number, number, number] = dark ? [242, 244, 248] : hexToRgb(cssVar("--foreground"));
   227→   227→
   228→   228→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   229→   229→
   230→   230→    // Fade (or clear) — this is what leaves the trails
   231→   231→    if (trailsRef.current) {
   232→   232→      ctx.fillStyle = rgba(bg, 0.055);
   233→   233→      ctx.fillRect(0, 0, w, h);
   234→   234→    } else {
   235→   235→      ctx.clearRect(0, 0, w, h);
   236→   236→      ctx.fillStyle = rgba(bg, 1);
   237→   237→      ctx.fillRect(0, 0, w, h);
   238→   238→    }
   239→   239→
   240→   240→    const cx = w / 2;
   241→   241→    const cy = h / 2;
   242→   242→    const scale = scaleRef.current;
   243→   243→
   244→   244→    // Analytic conic (dashed) — the exact solution, for contrast
   245→   245→    const orb = orbitRef.current;
   246→   246→    if (analyticRef.current && orb.bound) {
   247→   247→      const p = orb.h * orb.h; // semi-latus rectum (GM = 1)
   248→   248→      ctx.save();
   249→   249→      ctx.strokeStyle = rgba(dust, dark ? 0.55 : 0.6);
   250→   250→      ctx.lineWidth = 1;
   251→   251→      ctx.setLineDash([4, 6]);
   252→   252→      ctx.beginPath();
   253→   253→      for (let i = 0; i <= 240; i++) {
   254→   254→        const th = (i / 240) * Math.PI * 2;
   255→   255→        const rr = p / (1 + orb.e * Math.cos(th - orb.thetaP));
   256→   256→        if (rr <= 0 || rr > 60) continue;
   257→   257→        const X = cx + Math.cos(th) * rr * scale;
   258→   258→        const Y = cy - Math.sin(th) * rr * scale;
   259→   259→        if (i === 0) ctx.moveTo(X, Y);
   260→   260→        else ctx.lineTo(X, Y);
   261→   261→      }
   262→   262→      ctx.closePath();
   263→   263→      ctx.stroke();
   264→   264→
   265→   265→      // Periapsis / apoapsis ticks
   266→   266→      const rPeri = p / (1 + orb.e);
   267→   267→      const rApo = p / (1 - orb.e);
   268→   268→      const tick = (ang: number, rad: number, label: string) => {
   269→   269→        const X = cx + Math.cos(ang) * rad * scale;
   270→   270→        const Y = cy - Math.sin(ang) * rad * scale;
   271→   271→        ctx.setLineDash([]);
   272→   272→        ctx.strokeStyle = rgba(dust, 0.8);
   273→   273→        ctx.lineWidth = 1;
   274→   274→        ctx.beginPath();
   275→   275→        ctx.moveTo(X - 4, Y - 4);
   276→   276→        ctx.lineTo(X + 4, Y + 4);
   277→   277→        ctx.moveTo(X - 4, Y + 4);
   278→   278→        ctx.lineTo(X + 4, Y - 4);
   279→   279→        ctx.stroke();
   280→   280→        ctx.fillStyle = rgba(dust, 0.9);
   281→   281→        ctx.font = "10px ui-monospace, monospace";
   282→   282→        ctx.fillText(label, X + 7, Y - 5);
   283→   283→        ctx.setLineDash([4, 6]);
   284→   284→      };
   285→   285→      tick(orb.thetaP, rPeri, "P");
   286→   286→      tick(orb.thetaP + Math.PI, rApo, "A");
   287→   287→      ctx.restore();
   288→   288→    }
   289→   289→
   290→   290→    // Barycentre
   291→   291→    ctx.strokeStyle = rgba(dust, 0.7);
   292→   292→    ctx.lineWidth = 1;
   293→   293→    ctx.beginPath();
   294→   294→    ctx.moveTo(cx - 5, cy);
   295→   295→    ctx.lineTo(cx + 5, cy);
   296→   296→    ctx.moveTo(cx, cy - 5);
   297→   297→    ctx.lineTo(cx, cy + 5);
   298→   298→    ctx.stroke();
   299→   299→
   300→   300→    // Body positions: r1 = −(m2/M)·r, r2 = +(m1/M)·r, M = 1
   301→   301→    const rho = ratioRef.current;
   302→   302→    const m1 = 1 / (1 + rho);
   303→   303→    const m2 = rho / (1 + rho);
   304→   304→    const r = rRef.current;
   305→   305→    const x1 = cx - m2 * r.x * scale;
   306→   306→    const y1 = cy + m2 * r.y * scale; // canvas y is inverted
   307→   307→    const x2 = cx + m1 * r.x * scale;
   308→   308→    const y2 = cy - m1 * r.y * scale;
   309→   309→
   310→   310→    // Primary — warm starlight with a restrained glow
   311→   311→    const R1 = 8 + 7 * m1;
   312→   312→    const g1 = ctx.createRadialGradient(x1, y1, 0, x1, y1, R1 * 3);
   313→   313→    g1.addColorStop(0, rgba(star, 0.16));
   314→   314→    g1.addColorStop(1, rgba(star, 0));
   315→   315→    ctx.fillStyle = g1;
   316→   316→    ctx.fillRect(x1 - R1 * 3, y1 - R1 * 3, R1 * 6, R1 * 6);
   317→   317→    ctx.beginPath();
   318→   318→    ctx.arc(x1, y1, R1, 0, Math.PI * 2);
   319→   319→    ctx.fillStyle = `rgb(${star[0]}, ${star[1]}, ${star[2]})`;
   320→   320→    ctx.fill();
   321→   321→
   322→   322→    // Secondary — photon cyan
   323→   323→    const R2 = 3.5 + 9 * m2;
   324→   324→    const g2 = ctx.createRadialGradient(x2, y2, 0, x2, y2, R2 * 4);
   325→   325→    g2.addColorStop(0, rgba(photon, 0.28));
   326→   326→    g2.addColorStop(1, rgba(photon, 0));
   327→   327→    ctx.fillStyle = g2;
   328→   328→    ctx.fillRect(x2 - R2 * 4, y2 - R2 * 4, R2 * 8, R2 * 8);
   329→   329→    ctx.beginPath();
   330→   330→    ctx.arc(x2, y2, R2, 0, Math.PI * 2);
   331→   331→    ctx.fillStyle = `rgb(${photon[0]}, ${photon[1]}, ${photon[2]})`;

   345→   345→      const tipY = y2 + uy;
   346→   346→      ctx.strokeStyle = rgba(photon, 0.9);
   347→   347→      ctx.lineWidth = 1.5;
   348→   348→      ctx.beginPath();
   349→   349→      ctx.moveTo(x2, y2);
   350→   350→      ctx.lineTo(tipX, tipY);
   351→   351→      ctx.stroke();
   352→   352→      // arrowhead
   353→   353→      const ang = Math.atan2(uy, ux);
   354→   354→      ctx.beginPath();
   355→   355→      ctx.moveTo(tipX, tipY);
   356→   356→      ctx.lineTo(tipX - 8 * Math.cos(ang - 0.4), tipY - 8 * Math.sin(ang - 0.4));
   357→   357→      ctx.lineTo(tipX - 8 * Math.cos(ang + 0.4), tipY - 8 * Math.sin(ang + 0.4));

   359→   359→      ctx.fillStyle = rgba(photon, 0.9);
   360→   360→      ctx.fill();
   361→   361→    }
   362→   362→
   363→   363→    // Live readouts (direct DOM writes — no React re-render per frame)
   364→   364→    const epsNow = (v.x * v.x + v.y * v.y) / 2 - 1 / Math.hypot(r.x, r.y);
   365→   365→    const eNow = Math.sqrt(Math.max(0, 1 + 2 * epsNow * Math.pow(orbitRef.current.h, 2)));
   366→   366→    const aNow = epsNow < 0 ? -1 / (2 * epsNow) : Infinity;
   367→   367→    const dE = e0Ref.current !== 0 ? ((epsNow - e0Ref.current) / Math.abs(e0Ref.current)) * 100 : 0;
   368→   368→    const T = aNow < Infinity ? 2 * Math.PI * Math.pow(aNow, 1.5) : Infinity;
   369→   369→    if (readE.current) readE.current.textContent = eNow.toFixed(3);
   370→   370→    if (readA.current)
   371→   371→      readA.current.textContent = aNow < Infinity ? aNow.toFixed(3) : "—";
   372→   372→    if (readEps.current) readEps.current.textContent = epsNow.toFixed(3);
   373→   373→    if (readT.current)
   374→   374→      readT.current.textContent = T < Infinity ? `${T.toFixed(1)} τ` : "unbound";
   375→   375→    if (readDE.current) readDE.current.textContent = `${dE >= 0 ? "+" : ""}${dE.toFixed(3)} %`;
   376→   376→  }, []);
   377→   377→
   378→   378→  /* ---------- lifecycle ---------- */
   379→   379→
   380→   380→  useEffect(() => {
   381→   381→    const canvas = canvasRef.current;
   382→   382→    if (!canvas) return;
   383→   383→
   384→   384→    const resize = () => {
   385→   385→      const parent = canvas.parentElement;
   386→   386→      if (!parent) return;
   387→   387→      const rect = parent.getBoundingClientRect();
   388→   388→      if (rect.width < 2 || rect.height < 2) return;
   389→   389→      const dpr = Math.min(window.devicePixelRatio || 1, 2);
   390→   390→      const prevScale = scaleRef.current;
   391→   391→      sizeRef.current = { w: rect.width, h: rect.height, dpr };
   392→   392→      canvas.width = Math.round(rect.width * dpr);
   393→   393→      canvas.height = Math.round(rect.height * dpr);
   394→   394→      if (prevScale === 1) {
   395→   395→        resetSim(launchF);
   396→   396→      } else {
   397→   397→        // keep the orbit fitted on resize
   398→   398→        resetSim(launchF, false);
   399→   399→      }
   400→   400→    };
   401→   401→
   402→   402→    resize();
   403→   403→    resetSim(launchF);
   404→   404→
   405→   405→    const ro = new ResizeObserver(resize);
   406→   406→    ro.observe(canvas.parentElement ?? canvas);
   407→   407→
   408→   408→    const io = new IntersectionObserver(
   409→   409→      (entries) => {
   410→   410→        visibleRef.current = entries[0]?.isIntersecting ?? true;
   411→   411→      },
   412→   412→      { threshold: 0.05 }
   413→   413→    );
